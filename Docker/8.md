# =========================================================
# Dockerfile — nopCommerce (ASP.NET Core / .NET 9)
# Multi-stage build: SDK to build/publish, ASP.NET runtime to run
# =========================================================

# ---------- Stage 1: Build ----------
FROM mcr.microsoft.com/dotnet/sdk:9.0 AS build
WORKDIR /src

# Download & unpack the official nopCommerce release (source-included build)
# Swap this URL for the version you need
ARG NOP_VERSION=4.90.8
RUN apt-get update && apt-get install -y unzip wget \
    && wget https://github.com/nopSolutions/nopCommerce/releases/download/release-${NOP_VERSION}/nopCommerce_${NOP_VERSION}_Source.zip -O nopcommerce.zip \
    && unzip nopcommerce.zip -d . \
    && rm nopcommerce.zip

# Restore & publish the web project
WORKDIR /src/src/Presentation/Nop.Web
RUN dotnet restore
RUN dotnet publish -c Release -o /app/publish --no-restore

# ---------- Stage 2: Runtime ----------
FROM mcr.microsoft.com/dotnet/aspnet:9.0 AS runtime
WORKDIR /app

# libgdiplus is required for System.Drawing-based image processing (RichText/Gdip fix)
RUN apt-get update \
    && apt-get install -y --no-install-recommends libgdiplus \
    && rm -rf /var/lib/apt/lists/*

# Create required runtime directories
RUN mkdir -p /app/bin /app/logs

# Copy published output from build stage
COPY --from=build /app/publish .

# Run as non-root (mirrors www-data ownership pattern from bare-metal install)
RUN addgroup --system nopgroup && adduser --system --ingroup nopgroup nopuser \
    && chown -R nopuser:nopgroup /app
USER nopuser

ENV ASPNETCORE_ENVIRONMENT=Production
ENV ASPNETCORE_URLS=http://+:5000
ENV DOTNET_PRINT_TELEMETRY_MESSAGE=false

EXPOSE 5000

ENTRYPOINT ["dotnet", "Nop.Web.dll"]
